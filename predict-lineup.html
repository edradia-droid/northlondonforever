Match Details remains unchanged. // Replace the black/dark background that is connected to the photo border with // the exact pitch green used by this page. function nl4RemoveOdegaardBackground(img){ if(!img || !img.classList.contains('odegaard-photo') || img.dataset.nl4BgCleaned) return; img.dataset.nl4BgCleaned='1'; const run=()=>{ try{ const w=img.naturalWidth,h=img.naturalHeight; if(!w || !h)return; const canvas=document.createElement('canvas'); canvas.width=w;canvas.height=h; const ctx=canvas.getContext('2d',{willReadFrequently:true}); ctx.drawImage(img,0,0,w,h); const im=ctx.getImageData(0,0,w,h),d=im.data; const GREEN=[37,127,63]; const isBackground=(i)=>{ const r=d[i],g=d[i+1],b=d[i+2]; // Black/near-black background only. Do not alter Arsenal kit/skin colours. return Math.max(r,g,b)<85; }; const seen=new Uint8Array(w*h); const qx=new Int32Array(w*h),qy=new Int32Array(w*h); let head=0,tail=0; const push=(x,y)=>{ const k=y*w+x; if(seen[k])return; seen[k]=1;qx[tail]=x;qy[tail]=y;tail++; }; // Start from every border pixel that is actually dark. for(let x=0;x=w||ny>=h)continue; const nk=ny*w+nx,ni=nk*4; if(!seen[nk]&&isBackground(ni))push(nx,ny); } } ctx.putImageData(im,0,0); img.src=canvas.toDataURL('image/png'); img.dataset.nl4BgCleaned='done'; }catch(e){ console.warn('NL4 Ødegaard green-background treatment skipped',e); } }; if(img.complete)run();else img.addEventListener('load',run,{once:true}); } // Expose the cleanup handler globally because the rendered player uses an inline onload handler.
